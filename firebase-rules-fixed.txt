rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    function isAuthenticated() {
      return request.auth != null;
    }
    
    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }
    
    // Tomas collection - LECTURA PÚBLICA, ESCRITURA PROTEGIDA
    match /tomas/{tomaId} {
      // Permitir lectura a todos (incluso sin autenticación)
      allow read: if true;
      
      // Solo usuarios autenticados pueden crear
      allow create: if isAuthenticated() && 
                       request.resource.data.userId == request.auth.uid;
      
      // Solo el dueño puede actualizar
      allow update: if isAuthenticated() && 
                       resource.data.userId == request.auth.uid;
      
      // Solo el dueño puede borrar
      allow delete: if isAuthenticated() && 
                       resource.data.userId == request.auth.uid;
    }
    
    // Users collection - Lectura pública para ver perfiles
    match /users/{userId} {
      // Permitir lectura a todos para ver avatares y nombres
      allow read: if true;
      
      // Solo el dueño puede crear su perfil
      allow create: if isOwner(userId) &&
                       request.resource.data.uid == userId;
      
      // Solo el dueño puede actualizar su perfil
      allow update: if isOwner(userId);
      
      // Solo el dueño puede borrar su propio perfil
      allow delete: if isOwner(userId);
    }
  }
}
